COMPONENTES QUE SE RENDERIZAN Y ORGANIZACION

La pagina se organiza dentro de app/page.tsx. El orden actual es:
1. Portada, que incluye Reproductor.
2. ContadorAniversario.
3. Navegacion.
4. Una sola seccion condicional: LineaTiempo, Galeria o Cartitas.
5. TableroPostits, siempre renderizado fuera de la condicion de navegacion.
6. Pie de pagina.

La pagina mantiene activeSection como fuente unica para las pestañas y
openCard para conservar que carta se encuentra abierta al cambiar de seccion.
El tablero conserva internamente sus razones, imagenes y rotaciones.

Existen y se utilizan:
- app/page.tsx

components/MusicPlayerProvider.tsx.

'use client';

import { useEffect, useState } from 'react';

const targetDate = new Date('2026-10-19T20:00:00');

export default function ContadorAniversario() {
  const [timeLeft, setTimeLeft] = useState({
    days: 0,
    hours: 0,
    minutes: 0,
    seconds: 0,
    isFinished: false,
  });

  useEffect(() => {
    const calculateTime = () => {
      const now = new Date();
      const difference = targetDate.getTime() - now.getTime();

      if (difference <= 0) {
        setTimeLeft({ days: 0, hours: 0, minutes: 0, seconds: 0, isFinished: true });
        return;
      }

      const days = Math.floor(difference / (1000 * 60 * 60 * 24));
      const hours = Math.floor((difference / (1000 * 60 * 60)) % 24);
      const minutes = Math.floor((difference / 1000 / 60) % 60);
      const seconds = Math.floor((difference / 1000) % 60);

      setTimeLeft({ days, hours, minutes, seconds, isFinished: false });
    };

    calculateTime();
    const interval = setInterval(calculateTime, 1000);
    return () => clearInterval(interval);
  }, []);

  return (
    <section className="w-full bg-[#F4F5E8]/95 backdrop-blur-md border-2 border-dashed border-[#C8D5B5] rounded-3xl p-8 shadow-lg text-center relative overflow-hidden">
      <h2 className="text-base sm:text-lg font-serif text-[#3A4A32] mb-6 font-medium">
        {timeLeft.isFinished ? "¡Llegó nuestro aniversario! 🎉" : "Faltan para el 19 de Octubre (8:00 PM):"}
      </h2>
      
      <div className="grid grid-cols-2 md:grid-cols-4 gap-4 max-w-md mx-auto">
        <div className="bg-white/90 p-4 rounded-2xl shadow-sm border border-[#D4E0C4]">
          <span className="block text-2xl font-bold text-[#55684C]">{timeLeft.days}</span>
          <span className="text-[10px] uppercase tracking-wider text-[#6F8061]">Días</span>
        </div>
        <div className="bg-white/90 p-4 rounded-2xl shadow-sm border border-[#D4E0C4]">
          <span className="block text-2xl font-bold text-[#55684C]">{timeLeft.hours}</span>
          <span className="text-[10px] uppercase tracking-wider text-[#6F8061]">Horas</span>
        </div>
        <div className="bg-white/90 p-4 rounded-2xl shadow-sm border border-[#D4E0C4]">
          <span className="block text-2xl font-bold text-[#55684C]">{timeLeft.minutes}</span>
          <span className="text-[10px] uppercase tracking-wider text-[#6F8061]">Minutos</span>
        </div>
        <div className="bg-white/90 p-4 rounded-2xl shadow-sm border border-[#D4E0C4]">
          <span className="block text-2xl font-bold text-[#55684C]">{timeLeft.seconds}</span>
          <span className="text-[10px] uppercase tracking-wider text-[#6F8061]">Segundos</span>
        </div>
      </div>
    </section>
  );
}

'use client';

import { useRef, useState } from 'react';

export default function Reproductor() {
  const [isPlaying, setIsPlaying] = useState(false);
  const audioRef = useRef<HTMLAudioElement | null>(null);

  const togglePlay = () => {
    if (audioRef.current) {
      if (isPlaying) {
        audioRef.current.pause();
      } else {
        audioRef.current.play().catch((e) => console.log("Audio play error:", e));
      }
      setIsPlaying(!isPlaying);
    }
  };

  return (
    <>
      <audio ref={audioRef} src="/musica/cancion.mp3" loop />

      <button 
        onClick={togglePlay}
        className="flex items-center gap-3 bg-white/90 border border-[#C8D5B5] px-4 py-2.5 rounded-2xl shadow-sm hover:bg-[#F9FBF6] transition group"
      >
        <div className={`w-7 h-7 rounded-full bg-[#D4E0C4] flex items-center justify-center text-xs text-[#3A4A32] font-bold ${isPlaying ? 'animate-spin' : ''}`}>
          cass
        </div>
        <div className="text-left">
          <span className="block text-xs font-semibold text-[#3A4A32]">
            {isPlaying ? "Pausar música" : "Nuestra Canción 🎵"}
          </span>
          <span className="block text-[10px] text-[#6F8061]">
            {isPlaying ? "Reproduciendo..." : "Reproducir casete"}
          </span>
        </div>
      </button>
    </>
  );
}
